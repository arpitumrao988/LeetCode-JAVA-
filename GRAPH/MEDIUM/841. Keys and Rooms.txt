class Solution {
    public boolean canVisitAllRooms(List<List<Integer>> rooms) {
        int size = rooms.size();
        int[] visit = new int[size];
        Arrays.fill(visit , 0);
        Queue<Integer>q1 = new LinkedList<>();
        q1.offer(0);
        visit[0] =1;
        while(!q1.isEmpty())
        {
            int x = q1.poll();
            for(int i =0;i<rooms.get(x).size();i++)
            {
                int temp = rooms.get(x).get(i);
                if(visit[temp] == 0)
                {
                    visit[temp] = 1;
                    q1.offer(temp);
                }
            }
        }
        boolean flag = true;
        for(int i =0;i<size;i++)
        {
            if(visit[i] ==0 ) {flag = false; break;}
        }
        return flag;
    }
}
