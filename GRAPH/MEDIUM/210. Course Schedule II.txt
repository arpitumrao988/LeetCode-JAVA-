//210. Course Schedule II

class Solution {
    public int[] findOrder(int numCourses, int[][] prerequisites) {
        int[] inDegree = new int[numCourses];
        Arrays.fill(inDegree,0);
        int[] result = new int[numCourses];
        int k = 0;
        ArrayList<Integer>[] adjList = new ArrayList[numCourses];

        for (int i = 0; i < numCourses; i++) {
            adjList[i] = new ArrayList<>();
        }

        for (int[] p : prerequisites) {
            adjList[p[1]].add(p[0]);
            inDegree[p[0]]++;
        }

        Queue<Integer> q1 = new ArrayDeque<>();

        for (int i = 0; i < numCourses; i++) {
            if (inDegree[i] == 0) {
                q1.offer(i);
                result[k++]=i;
            }
        }
        if(q1.isEmpty())return new int[0];

        int completed = 0;

        while (!q1.isEmpty()) {
            int temp = q1.poll();
            completed++;

            for (int temp2 : adjList[temp]) {
                inDegree[temp2]--;

                if (inDegree[temp2] == 0) {
                    q1.offer(temp2);
                    result[k++]=temp2;
                }
            }
        }
        if(completed==numCourses)return result;
        return new int[0];

    }
}
