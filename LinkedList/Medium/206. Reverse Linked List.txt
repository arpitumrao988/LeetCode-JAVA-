class Solution {
    public ListNode reverseList(ListNode head) {
        if(head == null )return head;
        ListNode prev = null;
        ListNode current = head;
        ListNode Next = head.next;
        while(Next!=null)
        {
            current.next = prev;
            prev = current;
            current = Next;
            Next = Next.next;
        }
        current.next = prev;
        return current;
    }
}
