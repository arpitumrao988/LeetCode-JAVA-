class Result {

    /*
     * Complete the 'componentsInGraph' function below.
     *
     * The function is expected to return an INTEGER_ARRAY.
     * The function accepts 2D_INTEGER_ARRAY gb as parameter.
     */

    public static List<Integer> componentsInGraph(List<List<Integer>> gb) {
    // Write your code here
    
    int size = Integer.MIN_VALUE;
    for(List<Integer> ls:gb){
        size = Math.max(size,Math.max(ls.get(0), ls.get(1)));
    }
        boolean[] exists = new boolean[size + 1];
        ArrayList<Integer>[] adjList = new ArrayList[size+1];
        for(int i =0;i<size+1;i++)
        {
            adjList[i] = new ArrayList<>();
        }
        for(List<Integer>ls:gb){
            adjList[ls.get(0)].add(ls.get(1));
            adjList[ls.get(1)].add(ls.get(0));
            exists[ls.get(0)] = true;
            exists[ls.get(1)] = true;
        }
        int[] visit = new int[size+1];
        Arrays.fill(visit,0);
        visit[0] = 1;
        Queue<Integer>q1 = new LinkedList();
        int min = Integer.MAX_VALUE;
        int max = Integer.MIN_VALUE;
        int count=0;
        for(int i =0;i<size+1;i++){
            if(exists[i] && visit[i]==0){
                 count = 0;
            q1.offer(i);
            visit[i] = 1;
                while(!q1.isEmpty())
                {
                    int temp = q1.poll();
                    count++;
                    for(int j:adjList[temp])
                    {
                        if(visit[j]==0)
                        {
                            q1.offer(j);
                            visit[j] = 1;
                        }
                    }
                }
                 min = Math.min(min, count);
                max = Math.max(max,count);
            }
        }
        return Arrays.asList(min,max);
    }

}
