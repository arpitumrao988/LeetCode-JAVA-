         Queue<int[]>q1 = new ArrayDeque<>();

        }
        for(int i = 0;i<n;i++)
        {

        while(!q1.isEmpty())
        {

                {
                    int[] temp2 = q2.poll();
                    for(int i = 0;i<4;i++)
                    {
                        int new_r = temp2[0] +  directions[i][0];
                        int new_c = temp2[1] + directions[i][1];
                        if(new_r<m && new_r>=0 && new_c<n && new_c>=0 && grid[new_r][new_c]==0){q2.offer(new int[] {new_r,new_c});
                        grid[new_r][new_c] = 1;}
                    }
                }
            }
        }
        for(int[] i:grid)
        {
            System.out.println(Arrays.toString(i));
        }
        int closedIsland = 0;
        for(int i = 0;i<m;i++)
        {
            for(int j = 0;j<n;j++)
            {
                if(grid[i][j]==0)
                {
                    q1.offer(new int[] {i,j});
                    grid[i][j] = 1;
                    closedIsland++;
                    while(!q1.isEmpty())
                    {
                        int[] temp = q1.poll();
                        for(int k = 0;k<4;k++)
                        {
                            int new_r = temp[0] +  directions[k][0];
                            int new_c = temp[1] + directions[k][1];
                            if(grid[new_r][new_c]==0)q1.offer(new int[] {new_r,new_c});

class Solution {
    public int closedIsland(int[][] grid) {
        int m = grid.length;
        int n = grid[0].length;
        int closedIsland = 0;
        int[][] directions = {
            {1,0},{-1,0},{0,1},{0,-1}
        };
        Queue<int[]>q1 = new ArrayDeque<>();
        for(int i = 0;i<m;i++)
        {
            for(int j = 0;j<n;j++)
            {
                if(grid[i][j]==0)
                {
                    boolean isClosed = true;
                    q1.offer(new int[] {i,j});
                    grid[i][j] = 1;
                    while(!q1.isEmpty())
                    {
                        int[] temp = q1.poll();
                        if(temp[0]==0 || temp[0] == m-1 || temp[1]==0 || temp[1]==n-1)
                            {
                                isClosed = false;
                            }
                        for(int k = 0;k<4;k++)
                        {
                            int new_r = temp[0] +  directions[k][0];
                            int new_c = temp[1] + directions[k][1];
                            if(new_r<m && new_r>=0 && new_c<n && new_c>=0 &&grid[new_r][new_c]==0){
                                q1.offer(new int[] {new_r,new_c});
                                grid[new_r][new_c] = 1;   
                            }
                        }
                    }
                    if(isClosed)closedIsland++;
                }
            }
        }
        return closedIsland;
    }
}
