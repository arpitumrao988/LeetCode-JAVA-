class Solution {
    public String removeOuterParentheses(String s) {
        String result = "";
       Stack<Character>st = new Stack<>();
       for(int i = 0;i<s.length();i++)
       {
            if(s.charAt(i)=='('){ st.push('(');
            if(st.size()!=1) result += '(';}
            else{
                st.pop();
                if(!st.isEmpty()) result += ')' ;
            }   
       }
       return result;
    }
}
