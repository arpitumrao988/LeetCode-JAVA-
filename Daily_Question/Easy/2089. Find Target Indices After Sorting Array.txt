class Solution {
    public List<Integer> targetIndices(int[] nums, int target) {
        int startIndex = 0;
        int endIndex = -1;
        int n = nums.length;
        for(int i =0;i<n;i++)
        {
            if(nums[i]<target){
                startIndex++;
            }else if(nums[i]==target)endIndex++;
        }
        List<Integer>ls = new ArrayList<>();
        if(endIndex==-1)return ls;
        ls.add(startIndex);
        if(startIndex!=(endIndex+startIndex)){
            for(int i = startIndex+1;i<=endIndex+startIndex;i++){
            ls.add(i);}
            }
        return ls;
    }
}

//nlog(n)+log(n)+log(n)
class Solution {
    public List<Integer> targetIndices(int[] nums, int target) {
        Arrays.sort(nums);//nlog(n)
        int startIndex = binarySearch(nums,false,target);//log(n)
        int endIndex = binarySearch(nums,true,target);//log(n)
        List<Integer>ls = new ArrayList<>();
        for(int i = startIndex;i<=endIndex;i++)
        {
            ls.add(i);
        }
        return ls;
    }
    public int binarySearch(int[] arr,boolean flag,int key)
    {
        int start = 0;
        int end = arr.length-1;
        while(start<=end)
        {
            int mid = (start+end)/2;
            if(flag == true){
                if(arr[mid]<=key){
                    start = mid+1;
                }else {
                    end = mid-1;
                }
           }else{
                if(arr[mid]>=key){
                    end = mid-1;
                }else {
                    start = mid+1;
                }
            }
        }
        if(flag==true)return end;
        return start;
    }
}
