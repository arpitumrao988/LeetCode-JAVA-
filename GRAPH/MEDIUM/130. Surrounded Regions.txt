class Pair{
    int r;
    int c;
    Pair(int r, int c)
    {
        this.r = r;
        this.c = c;
    }
}
class Solution {
    public void solve(char[][] grid) {
        int[][] directions = {
            {1,0},{-1,0},{0,1},{0,-1}
        };
        int m = grid.length;
        int n = grid[0].length;
        //for upper edge
        for(int i =0;i<n;i++)
        {
            if(grid[0][i]=='O')
            {
                Queue<Pair>q1 = new LinkedList<>();
                q1.offer(new Pair(0,i));
                grid[0][i] = 'P';
                while(!q1.isEmpty())
                {
                    Pair temp = q1.poll();
                    int r = temp.r;
                    int c = temp.c;
                    for(int k = 0;k<4;k++)
                    {
                        int new_r = r + directions[k][0];
                        int new_c = c + directions[k][1];
                        if(new_r>=0 && new_c >= 0 && new_r<m && new_c<n && grid[new_r][new_c]=='O')
                        {
                            grid[new_r][new_c] = 'P';
                            q1.offer(new Pair(new_r,new_c));
                        }
                    }
                }
            }
        }
        //for lower edge
        for(int i =0;i<n;i++)
        {
            if(grid[m-1][i]=='O')
            {
                Queue<Pair>q1 = new LinkedList<>();
                q1.offer(new Pair(m-1,i));
                grid[m-1][i] = 'P';
                while(!q1.isEmpty())
                {
                    Pair temp = q1.poll();
                    int r = temp.r;
                    int c = temp.c;
                    for(int k = 0;k<4;k++)
                    {
                        int new_r = r + directions[k][0];
                        int new_c = c + directions[k][1];
                        if(new_r>=0 && new_c >= 0 && new_r<m && new_c<n && grid[new_r][new_c]=='O')
                        {
                            grid[new_r][new_c] = 'P';
                            q1.offer(new Pair(new_r,new_c));
                        }
                    }
                }
            }
        }
        //for left edge
        for(int i =0;i<m;i++)
        {
            if(grid[i][0]=='O')
            {
                Queue<Pair>q1 = new LinkedList<>();
                q1.offer(new Pair(i,0));
                grid[i][0] = 'P';
                while(!q1.isEmpty())
                {
                    Pair temp = q1.poll();
                    int r = temp.r;
                    int c = temp.c;
                    for(int k = 0;k<4;k++)
                    {
                        int new_r = r + directions[k][0];
                        int new_c = c + directions[k][1];
                        if(new_r>=0 && new_c >= 0 && new_r<m && new_c<n && grid[new_r][new_c]=='O')
                        {
                            grid[new_r][new_c] = 'P';
                            q1.offer(new Pair(new_r,new_c));
                        }
                    }
                }
            }
        }
        //for right edge
        for(int i =0;i<m;i++)
        {
            if(grid[i][n-1]=='O')
            {
                Queue<Pair>q1 = new LinkedList<>();
                q1.offer(new Pair(i,n-1));
                grid[i][n-1] = 'P';
                while(!q1.isEmpty())
                {
                    Pair temp = q1.poll();
                    int r = temp.r;
                    int c = temp.c;
                    for(int k = 0;k<4;k++)
                    {
                        int new_r = r + directions[k][0];
                        int new_c = c + directions[k][1];
                        if(new_r>=0 && new_c >= 0 && new_r<m && new_c<n && grid[new_r][new_c]=='O')
                        {
                            grid[new_r][new_c] = 'P';
                            q1.offer(new Pair(new_r,new_c));
                        }
                    }
                }
            }
        }
        for(char[] i :grid)
        {
            System.out.println(Arrays.toString(i));
        }
        int count = 0;
        for(int i =0;i<m;i++)
        {
            for(int j = 0;j<n;j++)
            {
                if(grid[i][j]=='O')grid[i][j] = 'X';
                if(grid[i][j]=='P')grid[i][j] = 'O';
            }
        }
        
    }
}
