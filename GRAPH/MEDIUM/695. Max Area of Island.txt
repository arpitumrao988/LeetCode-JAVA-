class Pair{
    int r;
    int c;
    Pair(int r, int c)
    {
        this.r = r;
        this.c = c;
    }
}
class Solution {
    public int maxAreaOfIsland(int[][] grid) {
        int[][] directions = {
            {1,0},{-1,0},{0,1},{0,-1}
        };
        int m = grid.length;
        int n = grid[0].length;
        int max_area = Integer.MIN_VALUE;
        int count = 0;
        for(int i =0;i<m;i++)
        {
            for(int j=0;j<n;j++)
            {
                count = 0;
                if(grid[i][j] == 1)
                { 
                  Queue<Pair>q1 = new LinkedList<>();
                    q1.offer(new Pair(i,j));
                    grid[i][j] = 0;
                    while(!q1.isEmpty())
                    {
                        Pair temp = q1.poll();
                        count++;
                        int r = temp.r;
                        int c = temp.c;
                        for(int k =0;k<4;k++)
                        {
                            int new_r = r + directions[k][0];
                            int new_c = c + directions[k][1];
                            if(new_c>=0 && new_r>=0 && new_c<n && new_r<m && grid[new_r][new_c]==1)
                            {
                              grid[new_r][new_c] = 0;
                              q1.offer(new Pair(new_r,new_c));  
                            }
                        }
                    }
                    
                }
                max_area = Math.max(max_area,count);
            }
        }
        return max_area;
    }
}
