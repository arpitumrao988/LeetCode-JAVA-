class Pair{
    int num;
    int count;
    Pair(int num, int count)
    {
        this.num = num;
        this.count = count;
    }
}
class Solution {
    public int snakesAndLadders(int[][] board) {
       int n = board.length;
       boolean[][] visit = new boolean[n][n];
       for(boolean[] arr :visit){
        Arrays.fill(arr,false);
       }
       Queue<Pair> q1 = new LinkedList<>();
       q1.offer(new Pair(1,0)); 
       int result = -1;
       visit[n-1][0] = true;
       while(!q1.isEmpty())
       {
            Pair temp = q1.poll();
            int square = temp.num;
            int dist = temp.count;
            if(square == n*n) {
                result = dist;
                break;
            }
            for(int i = 0;i<6;i++)
            {
                square++;
                if(square<=n*n){
                        int new_r = n-1-((square-1)/n);
                        int new_c = (square-1)%n;
                        if((n-1-new_r)%2==1)
                        {
                            new_c = n-1-new_c;
                        } 
                           
                            if(board[new_r][new_c]!=-1){
                                int r = n-1-((board[new_r][new_c]-1)/n);
                                int c = (board[new_r][new_c]-1)%n;
                                if((n-1-r)%2==1)
                                {
                                    c = n-1-c;
                                }
                                if(visit[r][c]!=true){
                                    q1.offer(new Pair(board[new_r][new_c],dist+1));
                                    visit[r][c]=true;}
                                }
                            else if(visit[new_r][new_c]!=true){
                                q1.offer(new Pair(square,dist+1));
                                visit[new_r][new_c]=true;
                            }    
                }
            }
       }
       return result;
    }
}
