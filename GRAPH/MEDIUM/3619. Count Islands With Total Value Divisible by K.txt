//3619. Count Islands With Total Value Divisible by K

class Solution {
    public int countIslands(int[][] grid, int k) {
        int m = grid.length;
        int n = grid[0].length;
        int[][] directions = {
            {1,0},{-1,0},{0,1},{0,-1}
        };
        int count = 0;
        for(int i = 0;i<m;i++)
        {
            for(int j = 0;j<n;j++)
            {
                if(grid[i][j] != 0)
                {
                    long sum = 0 ;
                    Queue<int[]> q1 = new ArrayDeque<>();
                    q1.offer(new int[]{i,j});
                    sum = grid[i][j];
                    grid[i][j] = 0;
                    while(!q1.isEmpty())
                    {
                        int[] temp = q1.poll();
                         for(int p = 0;p<4;p++)
                         {
                            int new_r = temp[0] +directions[p][0];
                            int new_c = temp[1] + directions[p][1];
                            if(new_r>=0 && new_r<m && new_c>=0 && new_c<n &&grid[new_r][new_c]!=0){
                                q1.offer(new int[]{new_r,new_c});
                                sum += grid[new_r][new_c];
                                grid[new_r][new_c] = 0;
                            }
                         } 
                    }
                    if(sum%k == 0)count++;
                }
            }
        }
        return count;
    }
}
