class Pair{
    int r ;
    int c ;
    public Pair(int r, int c)
    {
        this.r = r;
        this.c = c;
    }
}
class Solution {
    public int numIslands(char[][] grid) {
        int[][] directions = {
            {1,0},{-1,0},{0,1},{0,-1}
        };
        int count = 0;
        int m = grid.length;
        int n = grid[0].length;
        for(int  i =0;i<m;i++)
        {
            for(int j =0;j<n;j++)
            {
                if((grid[i][j]-48)==1)
                {
                    count++;
                    Queue<Pair>q1 = new LinkedList<>();
                    q1.offer(new Pair(i,j));
                    grid[i][j] = '0';
                    while(!q1.isEmpty())
                    {
                        Pair temp = q1.poll();
                        int r = temp.r;
                        int c = temp.c;
                        for(int k =0;k<4;k++)
                        {
                            int new_r = r + directions[k][0];
                            int new_c = c + directions[k][1];
                            if(new_r>=0 && new_c>=0 && new_r<m && new_c<n && grid[new_r][new_c]-48==1)
                            {
                                grid[new_r][new_c] = '0';
                                q1.offer(new Pair(new_r,new_c));
                            }
                        }
                    }
                }
            }
        }
        return count;
    }
}
